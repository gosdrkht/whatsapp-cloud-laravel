use LankaDev\WhatsApp\Facades\WhatsApp;

// Simple text message (local numbers auto-normalize when WHATSAPP_DEFAULT_COUNTRY_CODE=94)
WhatsApp::to('0771234567')->sendText('Hello from Sri Lanka!');

// Template message
WhatsApp::to('94771234567')->sendTemplate('order_confirmation', 'en', [
    'customer' => 'Kamal',
    'order_id' => 'ORD-1001',
]);

// Send a PDF invoice
WhatsApp::to('94771234567')->sendDocument(
    'https://example.com/invoice.pdf',
    'Invoice-1001.pdf'
);

// One-line OTP authentication template
WhatsApp::to('0771234567')->sendOtp('auth_otp_login', '482910');
use Illuminate\Notifications\Notification;
use LankaDev\WhatsApp\Notifications\WhatsAppMessage;

class OrderConfirmationNotification extends Notification
{
    public function via(object $notifiable): array
    {
        return ['whatsapp'];
    }

    public function toWhatsApp(object $notifiable): WhatsAppMessage
    {
        return WhatsAppMessage::template('order_confirmation', 'en', [
            'customer' => $notifiable->name,
            'order_id' => 'ORD-1001',
        ]);
    }
}

// Dispatch using Laravel's native notification system:
$user->notify(new OrderConfirmationNotification($order));
use Illuminate\Support\Facades\Event;
use LankaDev\WhatsApp\Events\MessageReceived;
use LankaDev\WhatsApp\Events\MessageFailed;

// GET & POST /whatsapp/webhook are auto-registered with raw-body
// X-Hub-Signature-256 verification and cache-backed idempotency.

Event::listen(MessageReceived::class, function (MessageReceived $event): void {
    $from = $event->message->from;
    $text = $event->message->text();
});

Event::listen(MessageFailed::class, function (MessageFailed $event): void {
    // Trigger local SMS fallback (Notify.lk / Text.lk) if WhatsApp fails
    logger()->warning('Delivery failed', [
        'error' => $event->status->firstErrorMessage(),
    ]);
});
use LankaDev\WhatsApp\Facades\WhatsApp;
use LankaDev\WhatsApp\Messages\MessagePayloadBuilder;

// Multi-tenant SaaS (per-branch POS / per-clinic HMS)
WhatsApp::forTenant($clinic)
    ->to('0771234567')
    ->sendText('Your lab report is ready for pickup.');

// Rate-paced bulk broadcast via Laravel Bus Batches
WhatsApp::broadcast(
    recipients: $subscribers->pluck('phone'),
    payloadFactory: fn (string $to) => MessagePayloadBuilder::template($to, 'promo_offer', 'en'),
    batchName: 'seasonal-promo',
    queue: 'whatsapp-bulk',
    rateLimitPerSecond: 20,
)->dispatch();
use LankaDev\WhatsApp\Facades\WhatsApp;

it('sends an order confirmation via WhatsApp', function (): void {

    WhatsApp::to('94771234567')->sendTemplate('order_confirmation', 'en', [

    ]);

Frequently Asked Questions

How do I get Meta WhatsApp Business Cloud API credentials?

Create a free Business app at Meta for Developers, add the WhatsApp product, and generate a Permanent System User token. Follow our complete Meta API Setup Guide.

Which Laravel and PHP versions are supported?

PHP 8.2, 8.3, and 8.4, and Laravel 11, 12, and 13 — tested on every push in GitHub Actions.

Can I use the client outside of Laravel?

Yes! Instantiate LankaDev\WhatsApp\Client\WhatsAppClient directly with WhatsAppConfig in any framework-independent PHP 8.2+ project.
